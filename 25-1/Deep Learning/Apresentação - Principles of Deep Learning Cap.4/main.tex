\begin{frame}{Reintroduzindo notação}

    Dataset com inputs $n_0$-dimensionais e com $N_D$ amostras: 

    $$
    D = \{x_{i,\alpha}\}_{i=1,\dots,n_0; \alpha=  1, \dots, N_D}
    $$

    Pré-ativações da 1ª camada são dadas por:

    $$
    z^{(1)}_{i,\alpha} \equiv z^{(1)}_i(x_\alpha) = b^{(1)}_i + \sum_{j=1}^{n_0} W^{(1)}_{ij} \, x_{j,\alpha}, \quad \text{para } i = 1, \dots, n_1.
    $$

\end{frame}

\begin{frame}{Reintroduzindo notação}
    Na inicialização, estabelecemos $b^{(1)},W^{(1)}$ independentemente distribuídas e tais que:
    
    $$
    \mathbb E[b_i^{(1)}] = \mathbb E[W_{i_1j_1}^{(1)}] = 0
    $$
    
    e
    
    \begin{align*}
    \mathbb E[b_i^{(1)}b_j^{(1)}] &= \delta_ij C_b^{(1)} \\
    \mathbb E[W_{i_1j_1}^{(1)}W_{i_2j_2}^{(1)}] &= \delta_{i_1i_2}\delta_{j_1j_2} \frac{C_W^{(1)}}{n_0}
    \end{align*}


\end{frame}


\begin{frame}{Objetivo}
    Estamos interessados em:

    $$
    p(z^{(1)}|D) = p(z^{(1)}(x_1),\dots,z^{(1)}(x_{N_D}))
    $$

\end{frame}

\begin{frame}
    \frametitle{4.1 - ``Wick this way'', derivação combinatória}

    Vemos primeiramente que:

    \begin{align*}
        \mathbb E[z_{i,\alpha}^{(1)}] = \mathbb E[b_i^{(1)} + \sum_{j=1}^{n_0}W_{ij}^{(1)}x_{j,\alpha_1}] \\
        = \mathbb E[b_i^{(1)}] + \sum_{j=1}^{n_0} \mathbb E[W_{ij}] x_{j,\alpha} = 0        
    \end{align*}

    dado que as distribuições tem média 0.

    É "fácil ver que" todos os correlatores de ordem ímpar de $p(z^{(1)}| D)$ são zerados, porque sempre sobra um $b^{(1)}$ ou $W^{(1)}$ em cada termo que zera a conta.

\end{frame}

\begin{frame}{4.1 - ``Wick this way'', derivação combinatória}
    \small
    Para um correlator de dois pontos:

    \begin{align*}
        \mathbb{E} \left[ z^{(1)}_{i_1;\alpha_1} \, z^{(1)}_{i_2;\alpha_2} \right] &= \mathbb{E} \left[
            \left( b^{(1)}_{i_1} + \sum_{j_1=1}^{n_0} W^{(1)}_{i_1 j_1} x_{j_1; \alpha_1} \right)
            \left( b^{(1)}_{i_2} +\sum_{j_2=1}^{n_0} W^{(1)}_{i_2 j_2} x_{j_2; \alpha_2} \right)
            \right]  \\ \Pause
            %
            &= \mathbb E [b_{i_1}b_{i_2}]  + \sum_{j_1}\mathbb E[b_{i_2}W_{i_1,j_1}]x_{j_1,\alpha_1} + \sum_{j_2}\mathbb E[b_{i_1}W_{i_2,j_2}]x_{j_2,\alpha_2}  \\
            %
            & \; + \mathbb E[(\sum_{j_1=1}^{n_0} W^{(1)}_{i_1 j_1} x_{j_1; \alpha_1})(\sum_{j_2=1}^{n_0} W^{(1)}_{i_2 j_2} x_{j_2; \alpha_2})] \Pause \\ 
            %
            &= \delta_{i_1 i_2}C_b^{(1)} + \sum_{j_1,j_2=1}^{n_0} \mathbb E[W_{i_1j_1}W_{i_2j_2}]x_{j,\alpha_1} x_{j,\alpha_2}  \Pause \\ 
            %
            &= \delta_{i_1 i_2} \left( C_b^{(1)} + C_W^{(1)} \frac{1}{n_0} \sum_{j=1}^{n_0} x_{j; \alpha_1} x_{j; \alpha_2} \right) 
    \end{align*}
\end{frame}

\begin{frame}
    \frametitle{4.1 - ``Wick this way'', derivação combinatória}

    Definimos a seguinte expressão como a \textbf{métrica de primeira camada}:
    $$
    G_{\alpha_1\alpha_2}^{(1)} := C_b^{(1)} + C_W^{(1)} \frac{1}{n_0} \sum_{j=1}^{n_0} x_{j; \alpha_1} x_{j; \alpha_2}
    $$ \pause

    Note que é uma função de $x_{\alpha_1}, x_{\alpha_2}$. Assim, nossa igualdade acima se traduz como: \pause

    $$
    \mathbb{E} \left[ z^{(1)}_{i_1;\alpha_1} \, z^{(1)}_{i_2;\alpha_2} \right] = \delta_{i_1i_2} G_{\alpha_1\alpha_2}^{(1)}
    $$

\end{frame}

\begin{frame}{4.1 - ``Wick this way'', derivação combinatória}
    \small
    Seguindo a mesma ideia:

    \begin{align*}
    \mathbb E[z^{(1)}_{i_1; \alpha_1} z^{(1)}_{i_2; \alpha_2} z^{(1)}_{i_3; \alpha_3} z^{(1)}_{i_4; \alpha_4}]
    = \delta_{i_1 i_2} \delta_{i_3 i_4} G^{(1)}_{\alpha_1 \alpha_2} G^{(1)}_{\alpha_3 \alpha_4}
    + \delta_{i_1 i_3} \delta_{i_2 i_4} G^{(1)}_{\alpha_1 \alpha_3} G^{(1)}_{\alpha_2 \alpha_4}\\
    + \delta_{i_1 i_4} \delta_{i_2 i_3} G^{(1)}_{\alpha_1 \alpha_4} G^{(1)}_{\alpha_2 \alpha_3} \Pause \\
    %
    = \mathbb{E} \left[ z^{(1)}_{i_1; \alpha_1} z^{(1)}_{i_2; \alpha_2} \right]
    \mathbb{E} \left[ z^{(1)}_{i_3; \alpha_3} z^{(1)}_{i_4; \alpha_4} \right] 
    + \mathbb{E} \left[ z^{(1)}_{i_1; \alpha_1} z^{(1)}_{i_3; \alpha_3} \right]
    \mathbb{E} \left[ z^{(1)}_{i_2; \alpha_2} z^{(1)}_{i_4; \alpha_4} \right] \\
    + \mathbb{E} \left[ z^{(1)}_{i_1; \alpha_1} z^{(1)}_{i_4; \alpha_4} \right]
    \mathbb{E} \left[ z^{(1)}_{i_2; \alpha_2} z^{(1)}_{i_3; \alpha_3} \right]
    \end{align*} \pause

    Lembrando de discussão anterior, isso significa que

    $$
    \mathbb E[z^{(1)}_{i_1; \alpha_1} z^{(1)}_{i_2; \alpha_2} z^{(1)}_{i_3; \alpha_3} z^{(1)}_{i_4; \alpha_4}]\big |_\text{connected} = 0
    $$
    de modo que a distribuição é Gaussiana. \pause

    De forma similar, podemos verificar que todos os correlatores conexos de ordem mais alta zeram, e portanto, podem ser gerados por uma Gaussiana.
\end{frame}

\begin{frame}
    \frametitle{4.1 - ``Wick this way'', derivação combinatória}

    Lembrando, uma distribuição Gaussiana multivariada $z$ de média $s$ e matriz de covariância $K$ pode ser escrita como

    $$
    p(z) = \frac{1}{\sqrt{|2\pi K|}} \exp\left[ - \frac 1 2 \sum_{\mu,\nu=1}^N (z-s)_\mu K  ^{\mu\nu}(z-s)_\nu \right] $$

    $$
    = \frac{1}{\sqrt{|2\pi K|}} \exp\left[ - \frac 1 2 (z-s)^T K^{-1} (z-s) \right]
    $$

    onde $K^{\mu\nu} = (K^{-1})_{\mu\nu}$.

\end{frame}

\begin{frame}
    \frametitle{4.1 - ``Wick this way'', derivação combinatória}

    Para escrever a distribuição de primeira camada, precisamos da inversa de  $\delta_{i_1i_2}G_{(1)}^{\alpha_1\alpha_2}$, para cada par de índices $i_1,i_2$, que satisfaça \pause

    $$
    \sum_{j=1}^{n_1} \sum_{\beta \in D} (\delta_{i_1 j} G^{\alpha_1 \beta}_{(1)})(\delta_{i_2 j} G_{\beta \alpha_2}^{(1)}) = \delta_{i_1i_2}\delta^{\alpha_1}_{\alpha_2}
    $$
    \pause

    $G^{\alpha \beta}_{(1)}$ denota as entradas da matriz $(G^{(1)})^{-1}_{\alpha \beta} \in \mathbb R^{N_D}$, com as métricas $G^{(1)}_{\alpha\beta}$ para duas entradas ${x_\alpha}, {x_\beta}$, satisfazendo

    $$
    \sum_{\beta \in D} G^{\alpha_1 \beta}_{(1)}G_{\beta \alpha_2}^{(1)} = \delta^{\alpha_1}_{\alpha_2}
    $$

\end{frame}

\begin{frame}
    \frametitle{4.1 - ``Wick this way'', derivação combinatória}

    Podemos, então, escrever a distribuição $p(z^{(1)}|D)$ na forma:
    \small
    $$
    p(z^{(1)}|D) = \frac{1}{Z} e^{-S(z^{(1)})}
    $$
    onde \pause $S$ é a *ação quadrática*
    $$
    S(z^{(1)}) = \frac{1}{2} \sum_{i=1}^{n_1} \sum_{\alpha_1,\alpha_2 \in D} z^{(1)}_{i;\alpha_1} G^{\alpha_1 \alpha_2}_{(1)} z^{(1)}_{i;\alpha_2}
    $$
    \pause
    $$
    \left(=\sum_{i,j=1}^{n_1} \sum_{\alpha_1,\alpha_2 \in D} z^{(1)}_{i;\alpha_1} \delta_{ij} G^{\alpha_1 \alpha_2}_{(1)} z^{(1)}_{j;\alpha_2} \right)
    $$
    \pause
    e $Z$ é a *função de partição*:
    $$
    Z = \int \left[ \prod_{i,\alpha} dz_{i,\alpha}^{(1)} \right] e^{-S(z^{(1)})} = |2 \pi G^{(1)}|^{\frac{n_1}{2}}
    $$
\end{frame}

\begin{frame}{4.1 - Hubbard-Stratonovich, derivação algébrica}
    
    Como visto no capítulo anterior, podemos expressar a distribuição na forma

    \footnotesize
    $$
    p(z \mid D) = \int \left[ \prod_j db_j \, p(b_j) \right] 
    \left[ \prod_{j,k} dW_{jk} \, p(W_{jk}) \right] 
    \prod_{j,\alpha} \delta \left( z_{j;\alpha} - b_i - \sum_j W_{jk} x_{j;\alpha} \right)
    $$
    \normalsize
    suprimindo por hora os sobrescritos $(1)$.
\end{frame}

\begin{frame}
    \frametitle{4.1 - Hubbard-Stratonovich, derivação algébrica}

    Por hipótese:
    $$
    p(b_j) = \frac{1}{\sqrt{2\pi C_b}} \exp(-\frac{b_j^2}{2C_b})
    $$

    $$
    p(W_{jk}) = \frac{1}{\sqrt{2\pi C_W/n_0}} \exp(-\frac{n_0 W_{jk}^2}{2C_W})
    $$

\end{frame}

    $$

\end{frame}

\begin{frame}{4.1 - Hubbard-Stratonovich, derivação algébrica}
    Ou seja, 
    $$
    -\frac{1}{2} \Lambda^T G \Lambda + i \Lambda^T z = -\frac{1}{2} \left[ (\Lambda - i G^{-1} z)^T G (\Lambda - i G^{-1} z) + z^TG^{-1}z\right].
    $$

    Em notação de somatório, para ficar igual ao livro
    \tiny
    \begin{gather*}
    -\frac{1}{2} \left[ (\Lambda - i G^{-1} z)^T G (\Lambda - i G^{-1} z) + z^TG^{-1}z\right] \\
    = - \frac{1}{2} \sum_{j, \alpha_1, \alpha_2} \left[
        G^{(1)}_{\alpha_1 \alpha_2} \left(
        \Lambda_{\alpha_1 j} - i \sum_{\beta_1} G_{(1)}^{\alpha_1 \beta_1} z^{(1)}_{j; \beta_1}
        \right)
        \left(
        \Lambda_{\alpha_2 j} - i \sum_{\beta_2} G_{(1)}^{\alpha_2 \beta_2} z^{(1)}_{j; \beta_2}
        \right) 
        + G_{(1)}^{\alpha_1 \alpha_2} z^{(1)}_{j; \alpha_1} z^{(1)}_{j; \alpha_2}
        \right]
    \end{gather*}
\end{frame}

\begin{frame}{4.1 - Hubbard-Stratonovich, derivação algébrica}
    \small
    Assim,
    \small
    \begin{gather*}
        p(z \mid D) =  \textcolor{teal}{\int \left[ \prod_{j,\alpha} \frac{d\Lambda_{j\alpha}}{2\pi} \right] 
        \exp \left[ (\Lambda - i G^{-1} z)^T G (\Lambda - i G^{-1} z)\right] }
        \exp\left[z^TG^{-1}z\right] \\
        = \textcolor{teal}{|2\pi G^{-1}|^{\frac{n_1}{2}}} \exp \left[z^TG^{-1}z\right] \\
        = \textcolor{teal}{\frac{1}{|2\pi G^{(1)}|^{\frac{n_1}{2}}}} \exp \left[- \frac{1}{2} \sum_{j, \alpha_1, \alpha_2} G_{(1)}^{\alpha_1 \alpha_2} z^{(1)}_{j; \alpha_1} z^{(1)}_{j; \alpha_2}\right]
    \end{gather*}

    voltando ao final o sobrescrito e a notação de somatório.

    \vspace{1em}
    \pause

    Ou seja, o mesmo termo que encontramos na seção ``Wick this way''.

\end{frame}

\subsection{Ação Gaussiana em ação}
\begin{frame}

\end{frame}

\begin{frame}
    \frametitle{4.1 - Ação Gaussiana em ação}

    \tiny

    \begin{align*}
        \mathbb{E} \left[ \sigma\left( z^{(1)}_{i_1;\alpha_1} \right) \sigma\left( z^{(1)}_{i_1;\alpha_2} \right) \right] \\
        = \int \left[ \prod_{i=1}^{n_1} \prod_{\alpha \in D} \frac{dz_{i;\alpha}}{\sqrt{|2\pi G^{(1)}|}} \right] \Pause
        % 
        \times  \exp\left( -\frac{1}{2} \sum_{j=1}^{n_1} \sum_{\beta_1, \beta_2 \in D} G^{(1)}_{\beta_1 \beta_2} z_{j;\beta_1} z_{j;\beta_2} \right)
        \sigma(z_{i_1;\alpha_1}) \sigma(z_{i_1;\alpha_2}) \Pause \\ 
        % 
        = \left\{ \prod_{i \neq i_1} \int \left[ \prod_{\alpha \in D} \frac{dz_{i;\alpha}}{\sqrt{|2\pi G^{(1)}|}} \right] 
        \exp \left[ -\frac{1}{2}\sum_{\beta_1, \beta_2 \in D} G^{(1)}_{\beta_1 \beta_2} z_{i;\beta_1} z_{i;\beta_2} \right] \right\}\Pause \\
        % 
        \times \int \left[ \prod_{\alpha \in D} \frac{dz_{i_1;\alpha}}{\sqrt{|2\pi G^{(1)}|}} \right]
        \exp\left( -\frac{1}{2} \sum_{\beta_1, \beta_2 \in D} G_{(1)}^{\beta_1 \beta_2} z_{i_1;\beta_1} z_{i_1;\beta_2} \right)
        \sigma(z_{i_1;\alpha_1}) \sigma(z_{i_1;\alpha_2})\Pause \\
        % 
        =\{1\} \times \left[ \int \prod_{\alpha \in D} \frac{dz_{i_1\alpha}}{\sqrt{|2\pi G^{(1)}|}}\right] 
        \exp\left( -\frac{1}{2} \sum_{\beta_1, \beta_2 \in D} G_{(1)}^{\beta_1 \beta_2} z_{i_1\beta_1} z_{i_1\beta_2} \right)
        \sigma(z_{i_1\alpha_1}) \sigma(z_{i_1\alpha_2})\Pause \\
        % 
        := \langle \sigma(z_{i_1\alpha_1})\sigma(z_{i_1\alpha_2}) \rangle_{G^{(1)}}.
    \end{align*}

\end{frame}

\begin{frame}{4.1 - Ação Gaussiana em ação}
    Com a expressão do final, reintroduzimos a notação:
    \tiny
    $$
    \left\langle F(z_{\alpha_1}, \ldots, z_{\alpha_m}) \right\rangle_g 
    \equiv 
    \displaystyle \int \left[\frac{\prod_{\alpha \in D} dz_\alpha}{\sqrt{|2\pi g|}} \right]
    \exp\left( -\frac{1}{2} \sum_{\beta_1, \beta_2 \in D} g^{\beta_1 \beta_2} z_{\beta_1} z_{\beta_2} \right)
    F(z_{\alpha_1}, \ldots, z_{\alpha_m})
    $$
    \normalsize
    \pause
    e denotamos ainda:

    $$
    \sigma_\alpha := \sigma(z_\alpha)
    $$
    \pause
    Assim, podemos escrever

    $$
    \mathbb{E} \left[ \sigma\left( z^{(1)}_{i_1;\alpha_1} \right) \sigma\left( z^{(1)}_{i_1;\alpha_2} \right) \right] = \langle \sigma_{\alpha_1} \sigma_{\alpha_2} \rangle_{G^{(1)}}
    $$

\end{frame}


\begin{frame}{4.1 - Ação Gaussiana em ação}
    É possível ainda generalizar para correlatores de mais pontos. Para todas as saídas do mesmo neurônio $i_1$:
    $$
    \mathbb{E} \left[ \sigma( z^{(1)}_{i_1;\alpha_1} ) \sigma( z^{(1)}_{i_1;\alpha_2})\sigma( z^{(1)}_{i_1;\alpha_3}) \sigma( z^{(1)}_{i_1;\alpha_4}) \right] = \langle \sigma_{\alpha_1} \sigma_{\alpha_2}\sigma_{\alpha_3}\sigma_{\alpha_4} \rangle_{G^{(1)}}
    $$
    fazendo as exatas mesmas etapas.
\end{frame}

\begin{frame}{4.1 - Ação Gaussiana em ação}
    Para $i_1\neq i_2$
    \tiny
    $$
    \mathbb{E} \left[
    \sigma\left( z^{(1)}_{i_1;\alpha_1} \right)
    \sigma\left( z^{(1)}_{i_1;\alpha_2} \right)
    \sigma\left( z^{(1)}_{i_2;\alpha_3} \right)
    \sigma\left( z^{(1)}_{i_2;\alpha_4} \right)
    \right]
    $$

    $$
    = \left\{ 
    \prod_{i \notin \{i_1, i_2\}} 
    \displaystyle \int \left[ \prod_{\alpha \in D} \frac{dz_{i;\alpha}}{\sqrt{|2\pi G^{(1)}|}} \right]
    \exp\left( -\frac{1}{2} \sum_{\beta_1, \beta_2 \in D} G_{(1)}^{\beta_1 \beta_2} z_{i;\beta_1} z_{i;\beta_2} \right)
    \right\}
    $$

    $$
    \times \displaystyle \int \left[ \frac{\prod_{\alpha \in D} dz_{i_1;\alpha}}{\sqrt{|2\pi G^{(1)}|}} \right]
    \exp\left( -\frac{1}{2} \sum_{\beta_1, \beta_2 \in D} G_{(1)}^{\beta_1 \beta_2} z_{i_1;\beta_1} z_{i_1;\beta_2} \right)
    \sigma(z_{i_1;\alpha_1}) \sigma(z_{i_1;\alpha_2})
    $$

    $$
    \times \displaystyle \int \left[ \frac{\prod_{\alpha \in D} dz_{i_2;\alpha}}{\sqrt{|2\pi G^{(1)}|}} \right]
    \exp\left( -\frac{1}{2} \sum_{\beta_1, \beta_2 \in D} G_{(1)}^{\beta_1 \beta_2} z_{i_2;\beta_1} z_{i_2;\beta_2} \right)
    \sigma(z_{i_2;\alpha_3}) \sigma(z_{i_2;\alpha_4})
    $$

    $$
    = \left\langle \sigma(z_{\alpha_1}) \sigma(z_{\alpha_2}) \right\rangle_{G^{(1)}}
    \left\langle \sigma(z_{\alpha_3}) \sigma(z_{\alpha_4}) \right\rangle_{G^{(1)}}
    $$


\end{frame}

\begin{frame}
    \frametitle{4.1 - Ação Gaussiana em ação}

    I.e.:
    \tiny
    $$
    \mathbb{E} \left[
    \sigma\left( z^{(1)}_{i_1;\alpha_1} \right)
    \sigma\left( z^{(1)}_{i_1;\alpha_2} \right)
    \sigma\left( z^{(1)}_{i_2;\alpha_3} \right)
    \sigma\left( z^{(1)}_{i_2;\alpha_4} \right)
    \right]
    = \left\langle \sigma(z_{\alpha_1}) \sigma(z_{\alpha_2}) \right\rangle_{G^{(1)}}
    \left\langle \sigma(z_{\alpha_3}) \sigma(z_{\alpha_4}) \right\rangle_{G^{(1)}}
    $$
    \normalsize
    Isso ilustra a independência entre as distribuições de neurônios diferentes.

\end{frame}

\subsection{Introdução}
\begin{frame}
\tableofcontents[currentsubsection]
\end{frame}

\begin{frame}{4.2 - Introdução}
    Lembrando, denotamos:
    \begin{equation}
        \sigma_{i;\alpha}^{(1)} := \sigma(z_{i;\alpha}^{(1)}).
    \end{equation}

    \pause
    As entradas da segunda camada são dadas por:
    \vspace{1em}
    
    \begin{equation}
        z^{(2)}_{i,\alpha} \equiv z^{(2)}_i(x_\alpha) = b^{(2)}_i + \sum_{j=1}^{n_1} W^{(2)}_{ij} \, \sigma_{j;\alpha}^{(1)}, \quad \text{para } i = 1, \dots, n_2.
    \end{equation}
\end{frame}

\begin{frame}{4.2 - Introdução}
    A distribuição conjunta da segunda e primeira camada pode ser dada por::
    \begin{equation}
        p(z^{(2)},z^{(1)} \mid D) = p(z^{(2)} \mid z^{(1)})p(z^{(1)} \mid D)
    \end{equation}

    \vspace{1em}
    1º: $p(z^{(1)} \mid D)$ - Seção 4.1; \pause

    \vspace{1em}
    2º: Como definido na seção 2:
    \tiny
    \begin{equation}
        p(z^{(2)} \mid z^{(1)}) = \int \left[\prod_i db_i^{(2)} p (b_i^{(2)}) \right] 
        \left[\prod_{i,j} dW_{ij}^{(2)} p (W_{ij}^{(2)}) \right] 
        \prod_{i,\alpha} \delta(z^{(2)}_i(x_\alpha) - b^{(2)}_i - \sum_{j=1}^{n_1} W^{(2)}_{ij} \, \sigma_{j;\alpha}^{(1)})
    \end{equation}
\end{frame}

\begin{frame}{4.2 - Introdução}
    Ainda, podemos calcular as probabilidades marginalizadas como:

    \begin{equation}
        p(z^{(2)} \mid D) = \int \left[\prod_{i;\alpha} z^{(1)} p(z^{(1)} \mid z^{(2)})\right]
    \end{equation}

\end{frame}

\subsection{Distribuição condicional de segunda camada}
\begin{frame}
\tableofcontents[currentsubsection]
\end{frame}

\begin{frame}{4.2 - Dist. condicional de 2ª camada} 
    Podemos calcular $p(z^{2} \mid z^{1})$ da exata mesma forma que $p(z^{(1)} \mid D)$: \pause
    \begin{itemize}
        \item índices $1 \to 2$; \pause
        \item entradas $x_{j;\alpha} \to \sigma_{j;\alpha}^{(1)}$. \pause
    \end{itemize}

    Desse modo, a probabilidade fica:
    \begin{equation}
        p(z^{(2)} \mid z^{(1)}) = \frac{1}{|2\pi \widehat G^{(2)}|^{\frac{n_2}{2}}} \exp \left[- \frac{1}{2} \sum_{j, \alpha_1, \alpha_2}^{n_2} \widehat G_{(2)}^{\alpha_1 \alpha_2} z^{(2)}_{j; \alpha_1} z^{(2)}_{j; \alpha_2}\right]
    \end{equation} 
    \pause
    onde definimos a \textbf{métrica \textit{estocástica} de segunda camada}:
    \begin{equation}
        \widehat G_{\alpha_1\alpha_2}^{(2)} := C_b^{(2)} + C_W^{(2)} \frac{1}{n_1} \sum_{j=1}^{n_1} \sigma_{j; \alpha_1} \sigma_{j; \alpha_2}
    \end{equation}
\end{frame}

\begin{frame}{4.2 - Dist. condicional de 2ª camada}
    Importante ressaltar que a variável $\widehat G^{(2)}$ é \textit{estocástica}: 
    \begin{equation}
        \widehat G_{\alpha_1\alpha_2}^{(2)} := C_b^{(2)} + C_W^{(2)} \frac{1}{n_1} \sum_{j=1}^{n_1} \sigma_{j; \alpha_1} \sigma_{j; \alpha_2}
    \end{equation}
    pois depende da variável aleatória $\sigma^{(1)} = \sigma(z^{(1)})$.
\end{frame}

\begin{frame}{4.2 - Dist. condicional de 2ª camada}
    Podemos então calcular seu valor esperado:
    \begin{align*}
        \mathbb E [\widehat G^{(2)_{\alpha_1 \alpha_2}}] &= C_b^{(2)} + C_W^{(2)} \frac{1}{n_1} \sum_{j=1}^{n_1} \mathbb E[\sigma_{j; \alpha_1} \sigma_{j; \alpha_2}] \Pause \\ 
        &= C_b^{(2)} + C_W^{(2)} \frac{1}{n_1} \sum_{j=1}^{n_1} \langle \sigma_{j; \alpha_1} \sigma_{j; \alpha_2} \rangle_{G^{(1)}} \Pause \\ 
        &:= G^{(2)}_{\alpha_1 \alpha_2}.
    \end{align*} 
    Chamamos o valor $G^{(2)}_{\alpha_1 \alpha_2}$ de \textbf{métrica média de 2ª camada.}
\end{frame}

\begin{frame}{4.2 - Dist. condicional de 2ª camada}
    Definimos ainda \textbf{flutuação de 2ª camada}:
    \begin{align*}
        \widehat{\Delta G}^{(2)}_{\alpha_1 \alpha_2} &:= \widehat G^{(2)}_{\alpha_1 \alpha_2} - G^{(2)}_{\alpha_1 \alpha_2} \Pause \\
        &= C_W^{(2)} \frac{1}{n_1} \sum_{j=1}^{n_1}(\sigma^{(1)}_{j;\alpha_1}\sigma^{(1)}_{j;\alpha_2} - \langle \sigma_{\alpha_1}\sigma_{\alpha_2}\rangle_{G^{(1)}})
    \end{align*}

    \pause
    
    Note que por construção, $\widehat{\Delta G}^{(2)}_{\alpha_1 \alpha_2} $ tem média 0 sobre a primeira camada de ativações:
    \begin{equation*} 
        \mathbb E[\widehat{\Delta G}^{(2)}_{\alpha_1 \alpha_2} ] = \mathbb E[\widehat G^{(2)}_{\alpha_1 \alpha_2}] - \mathbb E[ \mathbb E [\widehat G^{(2)}_{\alpha_1 \alpha_2}]] = 0
    \end{equation*}
\end{frame}

\begin{frame}{4.2 - Dist. condicional de 2ª camada}
    \scriptsize
    Estamos interessados no correlator de 2 pontos de $\widehat{\Delta G}^{(2)}_{\alpha_1 \alpha_2}$:
    \begin{align*}
        &\mathbb{E}\left[ \widehat{\Delta G}^{(2)}_{\alpha_1 \alpha_2} \widehat{\Delta G}^{(2)}_{\alpha_3 \alpha_4} \right] \Pause \\
        &= \left( \frac{C^{(2)}_W}{n_1} \right)^2 \sum_{j,k=1}^{n_1} 
        \mathbb{E} \left[ 
        \left( \sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2} - \mathbb{E}[\sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2}] \right)
        \left( \sigma^{(1)}_{k;\alpha_3} \sigma^{(1)}_{k;\alpha_4} - \mathbb{E}[\sigma^{(1)}_{k;\alpha_3} \sigma^{(1)}_{k;\alpha_4}] \right)
        \right] \Pause
    \end{align*}
    Temos:
    \begin{align*}
        &\mathbb{E} \left[ 
        \left( \sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2} - \mathbb{E}[\sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2}] \right)
        \left( \sigma^{(1)}_{k;\alpha_3} \sigma^{(1)}_{k;\alpha_4} - \mathbb{E}[\sigma^{(1)}_{k;\alpha_3} \sigma^{(1)}_{k;\alpha_4}] \right)
        \right] \Pause \\
        &=\mathbb{E} \left[\sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2} \sigma^{(1)}_{k;\alpha_3} \sigma^{(1)}_{k;\alpha_4} \right] 
        % %
        - \mathbb E\left[ \sigma^{(1)}_{k;\alpha_3} \sigma^{(1)}_{k;\alpha_4}\right] \mathbb{E}[\sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2}] \\
        % %    
        & \;\; -\mathbb E[\sigma^{(1)}_{j;\alpha_3} \sigma^{(1)}_{j;\alpha_4}] \mathbb{E}[\sigma^{(1)}_{k;\alpha_3} \sigma^{(1)}_{k;\alpha_4}] 
        % %
        + \mathbb E[\sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2}] \mathbb E[\sigma^{(1)}_{k;\alpha_3} \sigma^{(1)}_{k;\alpha_4}]
        \Pause \\
        & = \mathbb{E}\left[ \sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2} \sigma^{(1)}_{j;\alpha_3} \sigma^{(1)}_{j;\alpha_4} \right]
        - \mathbb{E}\left[ \sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2} \right]
        \mathbb{E}\left[ \sigma^{(1)}_{k;\alpha_3} \sigma^{(1)}_{k;\alpha_4}\right]
    \end{align*}

\end{frame}

\begin{frame}
    \frametitle{4.2 - Dist. condicional de 2ª camada}

    Note que, para $j\neq k$,
    \footnotesize
    \begin{equation*}
        \mathbb{E}\left[ \sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2} \sigma^{(1)}_{k;\alpha_3} \sigma^{(1)}_{k;\alpha_4} \right] = \mathbb{E}\left[ \sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2}\right] \left[\sigma^{(1)}_{j;\alpha_3} \sigma^{(1)}_{j;\alpha_4} \right]
    \end{equation*}

    \pause
    Com isso,
    \tiny 
    \begin{align*}
        &\left( \frac{C^{(2)}_W}{n_1} \right)^2 \sum_{j,k=1}^{n_1} 
        \mathbb{E} \left[ 
        \left( \sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2} - \mathbb{E}[\sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2}] \right)
        \left( \sigma^{(1)}_{k;\alpha_3} \sigma^{(1)}_{k;\alpha_4} - \mathbb{E}[\sigma^{(1)}_{k;\alpha_3} \sigma^{(1)}_{k;\alpha_4}] \right)
        \right] \Pause \\
        &= \left( \frac{C^{(2)}_W}{n_1} \right)^2 \sum_{j,k=1}^{n_1} 
        \left\{
        \mathbb{E}\left[ \sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2} \sigma^{(1)}_{k;\alpha_3} \sigma^{(1)}_{k;\alpha_4} \right]
        - \mathbb{E}\left[ \sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2} \right]
        \mathbb{E}\left[ \sigma^{(1)}_{k;\alpha_3} \sigma^{(1)}_{k;\alpha_4}\right]\right\}\Pause \\
        &= \left( \frac{C^{(2)}_W}{n_1} \right)^2 \sum_{j=1}^{n_1} 
        \left\{
        \mathbb{E}\left[ \sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2} \sigma^{(1)}_{j;\alpha_3} \sigma^{(1)}_{j;\alpha_4} \right]
        - \mathbb{E}\left[ \sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2} \right]
        \mathbb{E}\left[ \sigma^{(1)}_{j;\alpha_3} \sigma^{(1)}_{j;\alpha_4}\right]
        \right\} \Pause \\
        &= \left( \frac{C^{(2)}_W}{n_1} \right)^2 \sum_{j=1}^{n_1} 
        \left\{
        \mathbb{E}\left[ \sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2} \sigma^{(1)}_{j;\alpha_3} \sigma^{(1)}_{j;\alpha_4} \right]
        - \mathbb{E}\left[ \sigma^{(1)}_{j;\alpha_1} \sigma^{(1)}_{j;\alpha_2} \right]
        \mathbb{E}\left[ \sigma^{(1)}_{j;\alpha_3} \sigma^{(1)}_{j;\alpha_4}\right]
        \right\}\Pause \\
        &= \left( \frac{C^{(2)}_W}{n_1} \right)^2 \sum_{j=1}^{n_1} 
        \left\{
            \langle \sigma_{\alpha_1} \sigma_{\alpha_2} \sigma_{\alpha_3} \sigma_{\alpha_4} \rangle_{G^{(1)}} 
            - \langle \sigma_{\alpha_1} \sigma_{\alpha_2} \rangle_{G^{(1)}}
              \langle \sigma_{\alpha_3} \sigma_{\alpha_4} \rangle_{G^{(1)}}
        \right\} \Pause \\
        &= \frac{1}{n_1}\left(C^{(2)}_W\right)^2 
        \left[
            \langle \sigma_{\alpha_1} \sigma_{\alpha_2} \sigma_{\alpha_3} \sigma_{\alpha_4} \rangle_{G^{(1)}} 
            - \langle \sigma_{\alpha_1} \sigma_{\alpha_2} \rangle_{G^{(1)}}
              \langle \sigma_{\alpha_3} \sigma_{\alpha_4} \rangle_{G^{(1)}}
        \right]
    \end{align*}
\end{frame}

\begin{frame}
    \frametitle{4.2 - Dist. condicional de 2ª camada}

    O valor
    \small
    \begin{equation*}
        V_{(\alpha_1 \alpha_2)(\alpha_3 \alpha_4)}^{(2)}
        := \left(C^{(2)}_W\right)^2 
        \left[
            \langle \sigma_{\alpha_1} \sigma_{\alpha_2} \sigma_{\alpha_3} \sigma_{\alpha_4} \rangle_{G^{(1)}} 
            - \langle \sigma_{\alpha_1} \sigma_{\alpha_2} \rangle_{G^{(1)}}
              \langle \sigma_{\alpha_3} \sigma_{\alpha_4} \rangle_{G^{(1)}}
        \right]
    \end{equation*}
    é chamado de \textbf{vértice de quatro pontos} de 2ª camada. \pause Desse modo, podemos escrever
    \begin{equation*}
        \mathbb{E}\left[ \widehat{\Delta G}^{(2)}_{\alpha_1 \alpha_2} \widehat{\Delta G}^{(2)}_{\alpha_3 \alpha_4} \right] 
        = \frac{1}{n_1} V_{(\alpha_1 \alpha_2)(\alpha_3 \alpha_4)}^{(2)}.
    \end{equation*}
\end{frame}

\begin{frame}
    \frametitle{4.2 - Dist. condicional de 2ª camada}
    \textbf{Observações:}
    \begin{itemize}
        \item $V_{(\alpha_1 \alpha_2)(\alpha_3 \alpha_4)}^{(2)} == V(x_{\alpha_1},x_{\alpha_2};x_{\alpha_3},x_{\alpha_4})$;
        \item O vértice de quatro pontos é simétrico sobre $\alpha_1 \leftrightarrow \alpha_2$, $\alpha_3 \leftrightarrow \alpha_4$, $(\alpha_1,\alpha_2) \leftrightarrow (\alpha_3,\alpha_4)$. \pause I.e., \begin{itemize}
            \item $V_{(\alpha_1 \alpha_2)(\alpha_3 \alpha_4)}^{(2)} = V_{(\alpha_2 \alpha_1)(\alpha_3 \alpha_4)}^{(2)}$;
            \item $V_{(\alpha_1 \alpha_2)(\alpha_3 \alpha_4)}^{(2)} = V_{(\alpha_1 \alpha_2)(\alpha_4 \alpha_3)}^{(2)}$;
            \item $V_{(\alpha_1 \alpha_2)(\alpha_3 \alpha_4)}^{(2)} = V_{(\alpha_3 \alpha_4)(\alpha_1 \alpha_2)}^{(2)}$;
        \end{itemize}
    \item Quando $n_1 \gg 1$, $\frac{1}{n_1} V_{(\alpha_1 \alpha_2)(\alpha_3 \alpha_4)}^{(2)} \to 0$, de modo que a distribuição se aproxima de uma gaussiana.
    \end{itemize}
\end{frame}
